-- RACI MATRIX TABLE
-- Kopyahin lahat ng ito, i-paste sa Supabase SQL Editor, tapos pindutin ang Run.

CREATE TABLE IF NOT EXISTS raci_matrix (
  id          bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  process     text NOT NULL,
  dept        text NOT NULL,
  description text,
  responsible text NOT NULL,
  accountable text NOT NULL,
  consulted   text,
  informed    text,
  created_at  timestamptz DEFAULT now() NOT NULL
);

CREATE INDEX IF NOT EXISTS raci_matrix_dept_idx ON raci_matrix (dept);

ALTER TABLE raci_matrix DISABLE ROW LEVEL SECURITY;
GRANT ALL ON raci_matrix TO anon;
GRANT USAGE, SELECT ON ALL SEQUENCES IN SCHEMA public TO anon;
